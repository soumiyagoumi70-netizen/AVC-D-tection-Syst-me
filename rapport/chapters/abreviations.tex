\begin{description}
    \setlength{\itemsep}{0.25em}
    \item[API] \textit{Application Programming Interface} ; interface de programmation applicative.
    \item[AP] \textit{Average Precision} ; mesure synthétique de la courbe précision-rappel utilisée dans le benchmark.
    \item[AVC] Accident vasculaire cérébral.
    \item[EDA] \textit{Exploratory Data Analysis} ; analyse exploratoire des données.
    \item[\(F_2\)] Score \(F\) avec \(\beta=2\), accordant davantage de poids au rappel qu'à la précision.
    \item[FN] \textit{False Negative} ; faux négatif.
    \item[FP] \textit{False Positive} ; faux positif.
    \item[IMC] Indice de masse corporelle.
    \item[IQR] \textit{Interquartile Range} ; intervalle interquartile.
    \item[JSON] \textit{JavaScript Object Notation} ; format textuel d'échange de données.
    \item[KNN] \textit{K-Nearest Neighbors} ; méthode des \(k\) plus proches voisins.
    \item[MAR] \textit{Missing At Random} ; données manquantes au hasard conditionnellement aux variables observées.
    \item[MCAR] \textit{Missing Completely At Random} ; données manquantes complètement au hasard.
    \item[MNAR] \textit{Missing Not At Random} ; données manquantes non aléatoirement.
    \item[OOF] \textit{Out-of-Fold} ; prédiction hors pli de validation croisée.
    \item[PR-AUC] \textit{Precision-Recall Area Under the Curve} ; aire sous la courbe précision-rappel. Dans le benchmark, elle est estimée par l'\textit{Average Precision} (AP).
    \item[ROC-AUC] \textit{Receiver Operating Characteristic Area Under the Curve} ; aire sous la courbe ROC.
    \item[SMOTE] \textit{Synthetic Minority Over-sampling Technique} ; sur-échantillonnage synthétique de la classe minoritaire.
    \item[TN] \textit{True Negative} ; vrai négatif.
    \item[TP] \textit{True Positive} ; vrai positif.
    \item[XGBoost] \textit{Extreme Gradient Boosting} ; algorithme de gradient boosting extrême.
\end{description}
